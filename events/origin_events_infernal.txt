### Red Giant Supernova ###

planet_event = {
	id = infernals.9000
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		solar_system = {
			create_ambient_object = {
				type = "red_giant_supernova"
				play_animation_once = yes
				location = this
			}
			last_created_ambient_object = {
				set_location = {
					target = prev
					distance = 0
					angle = random
				}
			}
			
			#change to white dwarf 1 day later.. delay is for VFX
			star = {
				planet_event = {
					id = infernals.9001 days = 1
				}
			}
			
			#fade in nebulas 22 days later.. delay is for VFX
			star = {
				planet_event = {
					id = infernals.9002 days = 22
				}
			}
		}
	}
}

#changes to white dwarf
planet_event = {
	id = infernals.9001
	is_triggered_only = yes
	hide_window = yes
	immediate = {
		change_pc = pc_f_star
		solar_system = {
			set_star_class = sc_f
		}
		set_planet_size = 20
	}
}

#fade in nebulas
planet_event = {
	id = infernals.9002
	is_triggered_only = yes
	hide_window = yes
	immediate = {
		solar_system = {
			create_ambient_object = {
				type = "red_giant_supernova_nebula"
				location = this
				entity_scale_to_size = yes
				scale = 1.4
			}
			last_created_ambient_object = {
				set_location = {
					target = prev
					distance = 0
					angle = random
				}
			}
		}
	}
}
