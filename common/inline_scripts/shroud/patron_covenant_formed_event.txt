# equivalent to utopia.3310
country_event = {
	id = $EVENT_ID$
	title = utopia.3310.name
	picture = $PICTURE$
	show_sound = $SOUND$
	is_triggered_only = yes

	desc = {
		trigger = {
			if = {
				limit = {
					is_machine_empire = yes
				}
				text = $EVENT_ID$.desc.gestalt_machine
			}
			else_if = {
				limit = {
					is_robot_empire = yes
				}
				text = $EVENT_ID$.desc.individual_machine
			}
			else_if = {
				limit = {
					is_hive_empire = yes
				}
				text = $EVENT_ID$.desc.hive
			}
			else = {
				text = $EVENT_ID$.desc.default
			}
		}
	}

	show_sound = {
		trigger = { is_robot_empire = yes }
		sound = event_psionic_robot
	}
	show_sound = {
		trigger = { is_robot_empire = no }
		sound = event_psionic
	}

	option = {
		name = $EVENT_ID$.a.default
		trigger = {
			is_gestalt = no
		}
	}
	option = {
		name = $EVENT_ID$.a.gestalt
		trigger = {
			hidden_trigger = {
				is_gestalt = yes
			}
		}
	}
}
