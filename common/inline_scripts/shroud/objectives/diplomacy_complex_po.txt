country_event = {
	id = $EVENT_ID$
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		exists = from
		NOR = {
			has_country_flag = $FLAG_NAME$_@from
			from = {
				has_country_flag = $FLAG_NAME$_@prev
			}
		}
	}

	immediate = {
		set_country_flag = $FLAG_NAME$_@from
		if = {
			limit = { can_access_shroud = yes }
			add_patron_objective_counter = {
				counter = $PATRON_OBJECTIVE$
				amount = 1
			}
			add_patron_objective_counter = {
				counter = po_establish_diplomatic_pact
				amount = 1
			}
		}
		from = {
			if = {
				limit = { can_access_shroud = yes }
				add_patron_objective_counter = {
					counter = $PATRON_OBJECTIVE$
					amount = 1
				}
				add_patron_objective_counter = {
					counter = po_establish_diplomatic_pact
					amount = 1
				}
			}
		}
	}
}
