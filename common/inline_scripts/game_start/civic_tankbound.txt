# Tankbound Inital Setup
# Replace Farming Districts by Generator Districts

if = {
	limit = {
		exists = owner
		owner = {
			is_tankbound_empire = yes
		}
	}

	while = {
		limit = {
			check_variable = {
				which = farming_districts
				value > 0
			}
		}
		change_variable = {
			which = farming_districts
			value = -1
		}
		change_variable = {
			which = energy_districts
			value = 1
		}
	}
}
