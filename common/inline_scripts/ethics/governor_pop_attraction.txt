# Local Governor
modifier = {
	factor = 1.5
	from = {
		exists = leader
		leader = { is_$ETHIC$ = yes }
	}
}

modifier = {
	factor = 0.75
	from = {
		exists = leader
		leader = { is_$ETHIC_OPPOSED$ = yes }
	}
}

# Sector Governor (if no local governor)
modifier = {
	factor = 1.25
	from = {
		is_sector_capital = no
		NOT = { exists = leader }
		exists = sector
		sector.sector_capital = {
			exists = leader
			leader = { is_$ETHIC$ = yes }
		}
	}
}
