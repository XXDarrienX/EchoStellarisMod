empire_limit  = {
	base = 1
	modifier = {
		mult = owner.value:fe_building_cap|BUILDING|$BUILDING$|
		exists = owner
		owner = {
			NOR = {
				is_fallen_empire = yes
				has_ascension_perk = ap_cosmogenesis
				has_technology = $TECH$
			}
			has_country_flag = $FLAG$
		}
	}
	modifier = {
		mult = -1
		exists = owner
		owner = {
			OR = {
				is_fallen_empire = yes
				has_ascension_perk = ap_cosmogenesis
				has_technology = $TECH$
			}
		}
	}
}

planet_limit = {
	base = 1
	modifier = {
		mult = owner.value:fe_building_cap|BUILDING|$BUILDING$|
		exists = owner
		owner = {
			NOR = {
				is_fallen_empire = yes
				has_ascension_perk = ap_cosmogenesis
				has_technology = $TECH$
			}
			has_country_flag = $FLAG$
		}
	}
	modifier = {
		add = 2
		exists = owner
		owner = {
			has_technology = $TECH$
		}
	}
	modifier = {
		add = 3
		exists = owner
		owner = {
			has_ascension_perk = ap_cosmogenesis
		}
	}
	modifier = {
		mult = -1
		exists = owner
		owner = {
			is_fallen_empire = yes
		}
	}
}
