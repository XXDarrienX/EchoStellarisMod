provincial_administrative_supervision_attitude = {
	potential = {
		has_galactic_emperor = yes
		is_imperial_province = yes
	}

	option = {
		name = "provincial_administrative_supervision_cooperate"

		policy_flags = { provincial_administrative_supervision_cooperate }
		on_enabled = {
			galactic_emperor = {
				random_spynetwork = {
					limit = { target = { is_same_value = root } }
					add_modifier = { modifier = provincial_administrative_supervision_cooperate }
				}
			}
		}
		on_disabled = {
			galactic_emperor = {
				random_spynetwork = {
					limit = { target = { is_same_value = root } }
					remove_modifier = provincial_administrative_supervision_cooperate
				}
			}
		}

		AI_weight = {
			factor = 10
		}
	}
	option = {
		name = "provincial_administrative_supervision_resist"

		policy_flags = { provincial_administrative_supervision_resist }
		on_enabled = {
			galactic_emperor = {
				random_spynetwork = {
					limit = { target = { is_same_value = root } }
					add_modifier = { modifier = provincial_administrative_supervision_resist }
				}
			}
		}
		on_disabled = {
			galactic_emperor = {
				random_spynetwork = {
					limit = { target = { is_same_value = root } }
					remove_modifier = provincial_administrative_supervision_resist
				}
			}
		}

		AI_weight = {
			factor = 1
			modifier = {
				add = 20
				opinion_level = {
					who = galactic_emperor
					level < neutral
				}
			}
			modifier = {
				add = 100
				any_espionage_operation = {
					OR = {
						has_espionage_type = operation_disrupt_imperial_governance
						has_espionage_type = operation_weaken_galactic_empire
						has_espionage_type = operation_spark_rebellion
						has_espionage_type = operation_spark_rebellion_MECR
					}
				}
			}
		}
	}
}
