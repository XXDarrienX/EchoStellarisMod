politician_is_high_priest = {
	exists = owner
	OR = {
		owner = {
			has_valid_civic = civic_exalted_priesthood
		}
		has_building = building_citadel_of_faith
		has_building = building_capital_cathedral

	}
}

politician_is_science_director = {
	exists = owner
	OR = {
		owner = {
			has_valid_civic = civic_technocracy
		}
		has_building = building_institute
		has_building = building_vultaum_reality_computer
		has_building = building_fe_lab_1
		has_building = building_fe_lab_2
		has_building = building_silicon_valley
		has_building = building_space_agency
	}
}

bureaucrat_is_culture_worker = {
	OR = {
		has_building = building_artist_patron
		has_building = building_international_park
		has_building = building_central_memorial
		has_building = building_plentiful_traditions_tower_of_affinity
		num_zones = {
			type = zone_broken_shackles_memorial
			value >= 1
		}
	}
}
