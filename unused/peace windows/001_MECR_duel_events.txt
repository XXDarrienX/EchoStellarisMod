country_event = { #2833.计算分值
	id = MECR.2833
	is_triggered_only = yes
	hide_window = yes

	immediate = {
		event_target:pcwds_duel_loser = {
			set_variable = { which = pduel_score_round value = 0 }
			#从接壤星系开始，接壤星系为0
			every_system_within_border = {
				limit = { any_neighbor_system = { owner = { is_same_value = root } } }
				pduel_system_score_cal = yes
			}
			#接下来开始层层推进
			while = {
				limit = { #条件：有星系可以继续推进
					any_system_within_border = {
						has_star_flag = pduel_system_score_calculated
						any_neighbor_system = {
							owner = { is_same_value = event_target:pcwds_duel_loser }
							NOT = { has_star_flag = pduel_system_score_calculated }
						}
					}
				}
				#加一轮计数
				change_variable = { which = pduel_score_round value = 0.5 }
				#从上一轮完成的星系开始
				every_system_within_border = {
					limit = { has_star_flag = pduel_system_scoring_newest }
					remove_star_flag = pduel_system_scoring_newest
					every_neighbor_system = {
						limit = {
							owner = { is_same_value = event_target:pcwds_duel_loser }
							NOT = { has_star_flag = pduel_system_score_calculated }
						}
						pduel_system_score_cal = yes
					}
				}
			} #最终结果是从接壤星系开始，每推进一轮+1费
			every_system_within_border = {
				remove_star_flag = pduel_system_score_calculated
				remove_star_flag = pduel_system_scoring_newest
			}
			clear_variable = pduel_score_round
		}
		set_variable = { which = pdrs_marking_count value = 0 }
	}
}
