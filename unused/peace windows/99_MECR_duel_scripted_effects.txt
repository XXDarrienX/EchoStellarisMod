			create_pduel_take_system_poi = yes
			root = {

pduel_settlement_option_return_effect = {
	hidden_effect = {
		event_target:pdsm_$num$ = {
			remove_modifier = pduel_settlement_took_system
			remove_pduel_take_system_poi = yes
			root = {
				subtract_variable =  { which = pcwds_duel_win_score_used value = prev.duel_score_system }
				change_variable =  { which = pcwds_duel_win_score_left value = prev.duel_score_system }
			}
		}
		# 退还与已占领区断联的星系
		event_target:pcwds_duel_loser = {
			every_system_within_border = { #清除FLAG
				remove_star_flag = flag_pduel_recheck_link
				remove_star_flag = flag_pduel_recheck_link_newest
			}
			every_system_within_border = { #从接壤星系开始，追溯所有已取得的星系
				limit = {
					any_neighbor_system = { owner = { is_same_value = root } }
					has_modifier = pduel_settlement_took_system
					NOT = { has_star_flag = flag_pduel_recheck_link }
				}
				pduel_recheck_took_system_link = yes
			}
			while = { #接下来开始层层推进
				limit = { #条件：有星系可以继续推进
					any_system_within_border = {
						has_star_flag = flag_pduel_recheck_link
						any_neighbor_system = {
							owner = { is_same_value = event_target:pcwds_duel_loser }
							has_modifier = pduel_settlement_took_system
							NOT = { has_star_flag = flag_pduel_recheck_link }
						}
					}
				}
				#从上一轮完成的星系开始
				every_system_within_border = {
					limit = { has_star_flag = flag_pduel_recheck_link_newest }
					remove_star_flag = flag_pduel_recheck_link_newest
					every_neighbor_system = {
						limit = {
							owner = { is_same_value = event_target:pcwds_duel_loser }
							has_modifier = pduel_settlement_took_system
							NOT = { has_star_flag = flag_pduel_recheck_link }
						}
						pduel_recheck_took_system_link = yes
					}
				}
			} #最终结果是从接壤星系开始，给所有连接的已选中星系添加flag_pduel_recheck_link
			every_system_within_border = { #将无法追溯到的（断联星系）退还
				limit = {
					has_modifier = pduel_settlement_took_system
					NOT = { has_star_flag = flag_pduel_recheck_link }
				}
				remove_modifier = pduel_settlement_took_system
				remove_pduel_take_system_poi = yes
				root = {
					subtract_variable =  { which = pcwds_duel_win_score_used value = prev.duel_score_system }
					change_variable =  { which = pcwds_duel_win_score_left value = prev.duel_score_system }
				}
			}
			every_system_within_border = { #清除FLAG
				remove_star_flag = flag_pduel_recheck_link
				remove_star_flag = flag_pduel_recheck_link_newest
			}
		}
	}
}
pduel_settlement_auto_select_systems = {
	while = {
		count = 3
		# 1、计算虚拟价格
		set_variable = { which = pcwds_duel_win_score_left_ai_calc value = pcwds_duel_win_score_left }
		# 2、按价格立方权重随机选取所有相邻的星系（选得起的
		event_target:pcwds_duel_loser = {
			while = {
				limit = {
					any_system_within_border = {
						OR = {
							any_neighbor_system = { owner = { is_same_value = root } }
							any_neighbor_system = { has_modifier = pduel_settlement_took_system }
						}
						NOR = {
							has_modifier = pduel_settlement_took_system
							has_star_flag = pdrs_ai_would_like_to_take
						}
						root = { check_variable =  { which = pcwds_duel_win_score_left_ai_calc value >= prev.duel_score_system } }
					}
				}
				random_system_within_border = {
					limit = {
						OR = {
							any_neighbor_system = { owner = { is_same_value = root } }
							any_neighbor_system = { has_modifier = pduel_settlement_took_system }
						}
						NOR = {
							has_modifier = pduel_settlement_took_system
							has_star_flag = pdrs_ai_would_like_to_take
						}
						root = { check_variable =  { which = pcwds_duel_win_score_left_ai_calc value >= prev.duel_score_system } }
					}
					weights = {
						base = duel_score_system
						pow = 3
						add = -0.9
					}
					root = { subtract_variable =  { which = pcwds_duel_win_score_left_ai_calc value = prev.duel_score_system } }
					set_star_flag = pdrs_ai_would_like_to_take
				}
			}
		}
		# 3、选中所有目标星系
		event_target:pcwds_duel_loser = {
			while = {
