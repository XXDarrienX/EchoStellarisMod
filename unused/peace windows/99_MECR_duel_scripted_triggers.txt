pduel_is_protected_subject = {
	is_subject = yes
	any_agreement = {
		OR = {
			has_term_value = { term = joins_subject_wars value = joins_subject_wars_defensive }
			has_term_value = { term = joins_subject_wars value = joins_subject_wars_all }
		}
	}
}

pduel_precheck_duel_subject_check = {
	if = {
		limit = { is_subject = yes }
		custom_tooltip = {
			fail_text = "pduel_precheck_duel_we_are_subjects_under_one_overlord"
			NAND = {
				event_target:be_selected_as_pcwds_$num$ = { is_subject = yes }
				overlord = { is_same_value = event_target:be_selected_as_pcwds_$num$.overlord }
			}
		}
	}
}
pduel_precheck_duel_get_invitation = {
	if = {
		limit = {
			event_target:be_selected_as_pcwds_$num$ = {
				is_subject = yes
				any_agreement = {
					OR = {
						has_term_value = { term = joins_subject_wars value = joins_subject_wars_defensive }
						has_term_value = { term = joins_subject_wars value = joins_subject_wars_all }
					}
				}
			}
		}
		custom_tooltip = {
			fail_text = "pduel_precheck_duel_targets_overlord_get_invitation_in_five_years"
			event_target:be_selected_as_pcwds_$num$ = {
				overlord = { NOT = { has_country_flag = flag_refuse_pcwds_timer_common } }
			}
		}
	}
	custom_tooltip = {
		fail_text = "pduel_precheck_duel_get_invitation_in_five_years"
		event_target:be_selected_as_pcwds_$num$ = {
			NOT = { has_country_flag = flag_refuse_pcwds_timer_common }
		}
	}
}
pduel_precheck_duel_past_duel_between_us = {
	custom_tooltip = {
		fail_text = "pduel_precheck_duel_refuse_my_invitation_in_decade"
		event_target:be_selected_as_pcwds_$num$ = {
			NOT = { has_country_flag = flag_refuse_pcwds_timer_@root }
		}
	}
	custom_tooltip = {
		fail_text = "pduel_precheck_duel_we_ve_had_duel_in_decade"
		event_target:be_selected_as_pcwds_$num$ = {
			NOT = { has_country_flag = flag_accept_pcwds_timer_@root }
		}
	}
}
pduel_precheck_duel_arena = {
	custom_tooltip = {
		fail_text = "pduel_precheck_duel_arena_no_available"
		any_system = {
			NOT = {
				any_fleet_in_system = {
					owner = {
						OR = {
							is_hostile = root
							is_hostile = event_target:be_selected_as_pcwds_$num$
						}
					}
				}
			}
			NOR = { has_star_flag = pcwds_duel_arena }
			count_system_planet = { count > 1 }
			OR = {
				AND = {
					has_owner = yes
					owner = { is_same_value = event_target:be_selected_as_pcwds_$num$ }
					any_neighbor_system = {
						has_owner = yes
						owner = { is_same_value = root }
					}
				}
				AND = {
					has_owner = yes
					owner = { is_same_value = root }
					any_neighbor_system = {
						has_owner = yes
						owner = { is_same_value = event_target:be_selected_as_pcwds_$num$ }
					}
				}
			}
			OR = { #弃置不用的备选角斗场
				always = yes	
				AND = {
					OR = {
						has_owner = no
						owner = { is_galactic_custodian = yes }
						owner = { is_galactic_emperor = yes }
					}
					any_neighbor_system = {
						has_owner = yes
						owner = { is_same_value = event_target:be_selected_as_pcwds_$num$ }
					}
					any_neighbor_system = {
						has_owner = yes
						owner = { is_same_value = root }
					}
				}
				AND = {
					OR = {
						has_owner = no
						owner = { is_galactic_custodian = yes }
						owner = { is_galactic_emperor = yes }
					}
					distance_to_empire = {
						who = root
						distance < 3
						use_bypasses = no
					}
					distance_to_empire = {
						who = event_target:be_selected_as_pcwds_$num$
						distance < 3
						use_bypasses = no
					}
				}
				AND = {
					OR = {
						has_owner = no
						owner = { is_galactic_custodian = yes }
						owner = { is_galactic_emperor = yes }
					}
					any_system_megastructure = {
						exists = this
						OR = {
							is_megastructure_type = gateway_restored
							is_megastructure_type = gateway_final
						}
					}
				}
			}
		}
	}
}

pduel_settlement_option_take_trigger = {
	exists = event_target:pdsm_$num$
	event_target:pdsm_$num$ = {
		NOT = { has_modifier = pduel_settlement_took_system }
		OR = {
			any_neighbor_system = { owner = { is_same_value = root } }
			any_neighbor_system = { has_modifier = pduel_settlement_took_system }
		}
	}
}
pduel_settlement_option_take_allow = {
	custom_tooltip = {
		fail_text = "pcwds_duel_settlement_not_enough_score"
		event_target:pdsm_$num$ = {
			NOT = {
				check_variable = { which = duel_score_system value > root.pcwds_duel_win_score_left }
			}
		}
	}
}

pduel_settlement_option_return_trigger = {
	exists = event_target:pdsm_$num$
	event_target:pdsm_$num$ = {
		has_modifier = pduel_settlement_took_system
	}
}
pduel_settlement_option_return_allow = {
	if = {
		limit = {
			event_target:pdsm_$num$ = {
				any_neighbor_system = {
					has_modifier = pduel_settlement_took_system
					NOT = {
						any_neighbor_system = { owner = { is_same_value = root } }
					}
				}
			}
		}
		custom_tooltip = "pcwds_duel_settlement_may_return_other_systems"
	}
}
