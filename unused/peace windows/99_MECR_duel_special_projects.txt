special_project = {
	key = "pcwds_DUEL_ATTACKER_PROJECT"
	icon = "gfx/interface/icons/situation_log/situation_log_combat.dds"
	picture = GFX_evt_federation_fleet
	cost = 0
	days_to_research = 1					# days to research if cost is 0
	timelimit = 1800						# optional; time before it expires and fires on_fail

	location = yes

	event_scope = ship_event

	AI_wait_days = {
		base = 0
	}

	requirements = {
		leader = admiral
		shipclass_military = 1
	}

	on_success = {
		if = {
			limit = { fleet_size <= 200 }
			if = {
				limit = {
					NOR = {
						has_ship_owner_type = galactic_community
						has_ship_owner_type = federation
					}
				}
				fleet = {
					set_event_locked = yes
				}
				ship_event = {
					id = MECR.2812
				}
			}
			else = {
				prev = {
					enable_special_project = {
						name = pcwds_DUEL_ATTACKER_PROJECT
						location = from
						owner = prev
					}
				}
				ship_event = {
					id = MECR.2813
				}
			}
		}
		else = {
			prev = {
				enable_special_project = {
					name = pcwds_DUEL_ATTACKER_PROJECT
					location = from
					owner = prev
				}
			}
			ship_event = {
				id = MECR.2813
			}
		}
	}

	on_fail = {
	}
}
