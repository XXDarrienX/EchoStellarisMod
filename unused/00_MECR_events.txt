country_event = { #1023.【初次】帝国行省谈判：皇帝事件：循环判断谈判结果(1023.1024.)[1021.1023.]
	id = MECR.1023
	hide_window = yes
	is_triggered_only = yes
	
	immediate = {
		# 就联邦成员变动重新处理flag
		if = {
			limit = {
				any_federation_ally = {
					NOT = { has_country_flag = flag_become_province_asked }
				}
			}
			every_federation_ally = {
				limit = {
					NOT = { has_country_flag = flag_become_province_asked }
				}
				set_country_flag = flag_become_province_asked
				country_event = {
					id = MECR.1022
					days = 1
				}
			}
		}
		if = {
			limit = {
				any_country = {
					has_country_flag = flag_become_province_asked
					NOT = { is_in_federation_with = galactic_emperor }
				}
			}
			every_country = {
				limit = {
					has_country_flag = flag_become_province_asked
					NOT = { is_in_federation_with = galactic_emperor }
				}
				remove_country_flag = flag_become_province_asked
				remove_country_flag = flag_become_province_accepted
				remove_country_flag = flag_become_province_refused
			}
		}
		# 进行循环判定
		if = {
			limit = {
				any_federation_ally = {
					has_country_flag = flag_become_province_asked
					NOR = {
						has_country_flag = flag_become_province_accepted
						has_country_flag = flag_become_province_refused
					}
				}
			}
			country_event = {
				id = MECR.1023
				days = 2
			}
		}
		else = {
			country_event = {
				id = MECR.1024
			}
		}
	}
}
