country_event = { #2.每月给成员国添加扩张联邦舰队修正
	id = imp_forming.2
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_federation = yes
		federation = {
			has_federation_law = fed_law_independent_act
			OR = {
				has_federation_law = indact_federation_fleet_expansion_low
				has_federation_law = indact_federation_fleet_expansion_mid
				has_federation_law = indact_federation_fleet_expansion_high
			}
		}
	}

	immediate = {
		if = {
			limit = { federation = { has_federation_law = indact_federation_fleet_expansion_low } }
			add_modifier = {
				modifier = indact_federation_fleet_expansion
				multiplier = 1
				days = 31
			}
		}
		else_if = {
			limit = { federation = { has_federation_law = indact_federation_fleet_expansion_mid } }
			add_modifier = {
				modifier = indact_federation_fleet_expansion
				multiplier = 2
				days = 31
			}
		}
		else_if = {
			limit = { federation = { has_federation_law = indact_federation_fleet_expansion_high } }
			add_modifier = {
				modifier = indact_federation_fleet_expansion
				multiplier = 3
				days = 31
			}
		}
	}
}

country_event = { #20. 确认所有国家站队完毕，后续处理
	id = imp_forming.20
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			trigger = { any_country = { has_country_flag = flag_imp_forming_making_decision } } 
			country_event = { id = imp_forming.20 days = 1 }
		}
		else = {
			remove_global_flag = flag_MECR_emperor_20_on_fire
			if = { #如果有联邦通过了独立法案，作为统领联邦
				limit = {
					any_federation = {
						NOT = { has_federation_flag = special_federation }
						has_federation_law = fed_law_independent_act
					}
				}
				random_federation = {
					limit = {
						NOT = { has_federation_flag = special_federation }
						has_federation_law = fed_law_independent_act
					}
					weights = {
						factor = trigger:count_member
						modifier = {
							factor = 10
							any_member = { is_ai = no }
						}
					}
					save_global_event_target_as = leader_federation_against_imperium
					federation_leader = { save_global_event_target_as = leader_federation_against_imperium_leader }
					set_federation_flag = flag_federation_against_imperium
				}
				every_federation = {
					limit = {
						NOT = { has_federation_flag = special_federation }
						NOT = { is_same_value = event_target:leader_federation_against_imperium }
					}
					set_federation_flag = flag_federation_against_imperium
					if = {
						limit = {
							NOT = { any_member = { is_ai = no } }
							NOT = { has_federation_law = fed_law_independent_act }
						}
						set_federation_law = fed_law_independent_act
					}
				}
			}
			if = { #否则随机选择统领联邦
				limit = {
					NOT = { exists = event_target:leader_federation_against_imperium }
					any_federation = {
						NOT = { has_federation_flag = special_federation }
					}
				}
				random_federation = {
					limit = {
						NOT = { has_federation_flag = special_federation }
					}
					weights = {
						factor = trigger:count_member
						modifier = {
							factor = 10
							any_member = { is_ai = no }
						}
					}
					save_global_event_target_as = leader_federation_against_imperium
					federation_leader = { save_global_event_target_as = leader_federation_against_imperium_leader }
					set_federation_flag = flag_federation_against_imperium
					if = {
						limit = {
							NOT = { any_member = { is_ai = no } }
							NOT = { has_federation_law = fed_law_independent_act }
						}
						set_federation_law = fed_law_independent_act
					}
				}
				every_federation = {
					limit = {
						NOT = { has_federation_flag = special_federation }
						NOT = { is_same_value = event_target:leader_federation_against_imperium }
					}
					set_federation_flag = flag_federation_against_imperium
					if = {
						limit = {
							NOT = { any_member = { is_ai = no } }
							NOT = { has_federation_law = fed_law_independent_act }
						}
						set_federation_law = fed_law_independent_act
					}
				}
			}
			if = { #最糟糕的情况：没有任何联邦：向所有非帝国成员发送邀约建立联邦
				limit = { NOT = { exists = event_target:leader_federation_against_imperium } }
				set_global_flag = flag_imp_forming_no_federation_against_imperium
				every_playable_country = {
					limit = { is_galactic_community_member = no }
				}
			}

			export_trigger_value_to_variable = {
				trigger = count_playable_country
				parameters = { limit = { is_galactic_community_member = yes is_galactic_emperor = no } }
				variable = count_country_join_imperium
			}
			export_trigger_value_to_variable = {
				trigger = count_playable_country
				parameters = { limit = { is_galactic_community_member = no } }
				variable = count_country_leave_imperium
			}
			if = {
				limit = {
					check_variable = { which = count_country_leave_imperium value = 0 }
				}
				set_timed_global_flag = { flag = flag_imp_forming_result_best days = 1 }
			}
			else_if = {
				limit = {
					check_variable = { which = count_country_join_imperium value = 0 }
				}
				set_timed_global_flag = { flag = flag_imp_forming_result_worst days = 1 }
			}
			else_if = {
				limit = {
					check_variable = { which = count_country_join_imperium value <= count_country_leave_imperium }
				}
				set_timed_global_flag = { flag = flag_imp_forming_result_bad days = 1 }
			}
			else = {
				set_timed_global_flag = { flag = flag_imp_forming_result_good days = 1 }
			}
			clear_variable = count_country_join_imperium
			clear_variable = count_country_leave_imperium

			galactic_emperor = {
				country_event = {
					id = imp_forming.21
				}
			}
			every_playable_country = {
				limit = {
					is_galactic_emperor = no
				}
				country_event = {
					id = imp_forming.22
				}
			}
			if = { #没有反抗联邦，在所有非联邦成员中尝试建立
				limit = { has_global_flag = flag_imp_forming_no_federation_against_imperium }
				every_playable_country = {
					limit = {
						is_galactic_community_member = no
						has_federation = no
					}
					country_event = {
						id = imp_forming.24
						days = 14
					}
				}
			}
			else = { #否则，非联邦成员收到提示，寻求加入联邦
				every_playable_country = {
					limit = {
						is_galactic_community_member = no
						has_federation = no
					}
					country_event = {
						id = imp_forming.23
						days = 7
					}
				}
			}
		}
	}
}
