fed_law_independent_act = {
	potential = {
		NOT = { has_federation_flag = special_federation }
		OR = {
			has_federation_flag = flag_imperium_forming_federation_vote
			has_federation_flag = flag_federation_against_imperium
		}
	}
	allow = { always = yes }
	modifier =	{
	}
	on_enact = {
		if = {
			limit = {
				OR = {
					has_federation_law = centralization_high
					has_federation_law = centralization_very_high
				}
			}
			custom_tooltip = "fed_law_independent_act_effect_high_centralization"
		}
		else = {
			custom_tooltip = "fed_law_independent_act_effect"
		}
		hidden_effect = {
			remove_federation_flag = flag_imperium_forming_federation_lag
			federation_leader = {
				country_event = {
					id = imp_forming.1
					days = 1
				}
			}
		}
	}
	cohesion_growth = 5
	ai_weight = {
		base = 1
		modifier = {
			desc = federation_acceptance_president
			add = 25
			is_federation_leader = yes
		}
		modifier = { #全部联邦成员非玩家时	-15
			desc = federation_acceptance_imp_no_player_federation
			add = -15
			is_ai = yes
			NOT = { any_federation_ally = { is_ai = no } }
		}
		complex_trigger_modifier = {	#对主席好感度*0.1
			potential = { is_federation_leader = no }
			desc = federation_acceptance_imp_relative_opinion_leader
			trigger = opinion
			parameters = { who = federation.leader }
			mode = add
			mult = 0.1
		}
		complex_trigger_modifier = {	#对皇帝好感度*-0.3
			desc = federation_acceptance_imp_relative_opinion_emperor
			trigger = opinion
			parameters = { who = galactic_emperor }
			mode = add
			mult = -0.3
		}
		# 相对实力
		modifier = {
			desc = federation_acceptance_imp_relative_power
			add = -10
			relative_power = { who = galactic_emperor category = all value = pathetic }
		}
		modifier = {
			desc = federation_acceptance_imp_relative_power
			add = 20
			relative_power = { who = galactic_emperor category = all value = inferior }
		}
		modifier = {
			desc = federation_acceptance_imp_relative_power
			add = 50
			relative_power = { who = galactic_emperor category = all value = equivalent }
		}
		modifier = {
			desc = federation_acceptance_imp_relative_power
			add = 100
			relative_power = { who = galactic_emperor category = all value = superior }
		}
		modifier = {
			desc = federation_acceptance_imp_relative_power
			add = 250
			relative_power = { who = galactic_emperor category = all value = overwhelming }
		}
		# 联邦盟友的相对实力
		modifier = {
			desc = federation_acceptance_imp_relative_power_federation
			add = -30
			NOT = {
				any_federation_ally = {
					relative_power = { who = galactic_emperor category = all value > pathetic }
				}
			}
		}
		modifier = {
			desc = federation_acceptance_imp_relative_power_federation
			add = 30
			any_federation_ally = {
				relative_power = { who = galactic_emperor category = all value >= equivalent }
			}
		}
	}
}
