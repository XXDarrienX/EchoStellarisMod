# imperial_forming
flagging_the_countries_when_forming_imperium = {
	every_playable_country = { #皇帝的附庸
		limit = {
			is_subject = yes
			overlord = { is_galactic_emperor = yes }
		}
		set_timed_country_flag = {
			flag = flag_pre_imperium_forming_emperors_subject
			days = 60
		}
		set_country_flag = flag_pre_imperium_forming_flagged
	}
	every_playable_country = { #皇帝的非附庸盟友
		limit = {
			NOT = { has_country_flag = flag_pre_imperium_forming_flagged }
			is_in_federation_with = galactic_emperor
			is_subject = no
		}
		set_timed_country_flag = {
			flag = flag_pre_imperium_forming_emperors_alliances
			days = 60
		}
		set_country_flag = flag_pre_imperium_forming_flagged
	}
	every_playable_country = { #皇帝的盟友的附庸
		limit = {
			NOT = { has_country_flag = flag_pre_imperium_forming_flagged }
			is_subject = yes
			overlord = { is_in_federation_with = galactic_emperor }
		}
		set_timed_country_flag = {
			flag = flag_pre_imperium_forming_emperors_alliances_subject
			days = 60
		}
		set_country_flag = flag_pre_imperium_forming_flagged
	}
	every_playable_country = { #联邦非附庸成员(内)（如果是宗主，额外标志）
		limit = {
			NOT = { has_country_flag = flag_pre_imperium_forming_flagged }
			is_galactic_community_member = yes
			has_federation = yes
			is_subject = no
		}
		set_timed_country_flag = {
			flag = flag_pre_imperium_forming_federation_member
			days = 60
		}
		if = {
			limit = { is_overlord = yes }
			set_timed_country_flag = {
				flag = flag_pre_imperium_forming_federation_member_is_overlord
				days = 60
			}
		}
		set_country_flag = flag_pre_imperium_forming_flagged
	}
	every_playable_country = { #联邦非附庸成员(外)（如果是宗主，额外标志）
		limit = {
			NOT = { has_country_flag = flag_pre_imperium_forming_flagged }
			is_galactic_community_member = no
			has_federation = yes
			is_subject = no
		}
		set_timed_country_flag = {
			flag = flag_pre_imperium_forming_federation_member_outsider
			days = 60
		}
		if = {
			limit = { is_overlord = yes }
			set_timed_country_flag = {
				flag = flag_pre_imperium_forming_federation_member_outsider_is_overlord
				days = 60
			}
		}
		set_country_flag = flag_pre_imperium_forming_flagged
	}
	every_playable_country = { #附庸(内)(宗主内/宗主联邦成员外)
		limit = {
			NOT = { has_country_flag = flag_pre_imperium_forming_flagged }
			is_galactic_community_member = yes
			is_subject = yes
			overlord = {
				OR = {
					is_galactic_community_member = yes
					AND = {
						is_galactic_community_member = no
						has_federation = yes
					}
				}
			}
		}
		set_timed_country_flag = {
			flag = flag_pre_imperium_forming_subject
			days = 60
		}
		overlord = {
			set_timed_country_flag = {
				flag = flag_pre_imperium_forming_overlord_of_@prev
				days = 60
			}
		}
		set_country_flag = flag_pre_imperium_forming_flagged
	}
	every_playable_country = { #附庸(内)(宗主外)
		limit = {
			NOT = { has_country_flag = flag_pre_imperium_forming_flagged }
			is_galactic_community_member = yes
			is_subject = yes
			overlord = { is_galactic_community_member = no }
		}
		set_timed_country_flag = {
			flag = flag_pre_imperium_forming_outsiders_subject
			days = 60
		}
		overlord = {
			set_timed_country_flag = {
				flag = flag_pre_imperium_forming_overlord_of_@prev
				days = 60
			}
		}
		set_country_flag = flag_pre_imperium_forming_flagged
	}
	every_playable_country = { #附庸(外)(宗主内/宗主联邦成员外)
		limit = {
			NOT = { has_country_flag = flag_pre_imperium_forming_flagged }
			is_galactic_community_member = no
			is_subject = yes
			overlord = {
				OR = {
					is_galactic_community_member = yes
					AND = {
						is_galactic_community_member = no
						has_federation = yes
					}
				}
			}
		}
		set_timed_country_flag = {
			flag = flag_pre_imperium_forming_subject_outsider
			days = 60
		}
		overlord = {
			set_timed_country_flag = {
				flag = flag_pre_imperium_forming_overlord_of_@prev
				days = 60
			}
		}
		set_country_flag = flag_pre_imperium_forming_flagged
	}
	every_playable_country = { #宗主(内)
		limit = {
			NOT = { has_country_flag = flag_pre_imperium_forming_flagged }
			is_galactic_community_member = yes
			is_overlord = yes
		}
		set_timed_country_flag = {
			flag = flag_pre_imperium_forming_overlord
			days = 60
		}
		set_country_flag = flag_pre_imperium_forming_flagged
	}
	every_playable_country = { #其他(外)
		limit = {
			NOT = { has_country_flag = flag_pre_imperium_forming_flagged }
			is_galactic_community_member = no
		}
		set_timed_country_flag = {
			flag = flag_pre_imperium_forming_outsider
			days = 60
		}
		set_country_flag = flag_pre_imperium_forming_flagged
	}
	every_playable_country = { #其他(内)
		limit = {
			NOT = { has_country_flag = flag_pre_imperium_forming_flagged }
		}
		set_timed_country_flag = {
			flag = flag_pre_imperium_forming_regular
			days = 60
		}
		set_country_flag = flag_pre_imperium_forming_flagged
	}
	every_federation = { #联邦打上flag
		limit = { NOT = { any_member = { is_galactic_emperor = yes } } }
		set_federation_flag = flag_imperium_forming_federation_vote
		set_timed_federation_flag = {
			flag = flag_imperium_forming_federation_lag
			days = 14
		}
	}
	every_country = {
		limit = { has_country_flag = flag_pre_imperium_forming_flagged }
		remove_country_flag = flag_pre_imperium_forming_flagged
	}
	set_timed_global_flag = {
		flag = flag_imperium_forming_timer_30
		days = 29
	}
	galactic_emperor = {
		country_event = { id = imp_forming.20 days = 15 }
	}
}
move_subject_to_emperor = {
	set_subject_of = {
		who = galactic_emperor
		preset = preset_imperial_member_state
	}
}
subject_betray_to_emperor_get = {
	every_country = {
		limit = { has_country_flag = flag_pre_imperium_forming_overlord_of_@prev }
		add_casus_belli = {
			type = cb_subjugation
			who = prev
			days = -1
		}
		add_opinion_modifier = {
			who = prev
			modifier = opinion_imp_forming_subject_betray_us
		}
	}
}
become_member_state = {
	set_subject_of = {
		who = galactic_emperor
		preset = preset_imperial_member_state
	}
}
